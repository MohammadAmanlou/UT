
\section*{شماره سوال 1}

دو تكه كد به زبان \lr{Go} بنويسيد كه دو \lr{goroutine} متفاوت آن‌ها را اجرا مي‌كنند و در نتيجه انحصار متقابل اشتباه دچار بن‌بست شده باشند

\bigskip

بن‌بست (\lr{deadlock}) وضعیتی است که در آن دو یا چند \lr{goroutine} هر کدام قفلی را در اختیار دارند و منتظر آزاد شدن قفل دیگری هستند در نتیجه هیچ‌یک پیشرفت نمی‌کنند و برنامه متوقف می‌شود کانال بدون بافر (\lr{unbuffered} \lr{channel}) کانالی است که هیچ فضای میانی برای نگهداری داده ندارد وقتی یک \lr{goroutine} مقداری به این کانال می‌فرستد تا زمانی که \lr{goroutine} دیگری آن مقدار را دریافت نکند ارسال بلوکه می‌شود و بالعکس اگر دریافت درخواست شود اما مقداری ارسال نشده باشد دریافت بلوکه می‌شود

\bigskip

\textbf{مثال اول : دو قفل (\lr{Mutex}) با ترتیب معکوس:}

در این مثال دو قفل مجزا به نام‌های \lr{muA} و \lr{muB} تعریف کرده‌ایم دو \lr{goroutine} هم‌زمان اجرا می‌شوند \lr{goroutine} نخست ابتدا قفل \lr{muA} را می‌گیرد و سپس پس از مکث کوتاهی دنبال قفل \lr{muB} می‌گردد \lr{goroutine} دوم ابتدا قفل \lr{muB} را می‌گیرد و بعد تلاش می‌کند قفل \lr{muA} را به دست آورد هر \lr{goroutine} قفل اولیه را در اختیار دارد و منتظر آزاد شدن قفل دوم است بنابراین هر دو در حالت بن‌بست می‌مانند و برنامه دیگر جلو نمی‌رود

\medskip

\begin{latin}
\begin{lstlisting}[language=Go]
var muA sync.Mutex
var muB sync.Mutex
var wg  sync.WaitGroup

func main() {
    wg.Add(2)
    go func() {
        defer wg.Done()
        fmt.Println("goroutine 1: locked muA")
        muA.Lock()
        time.Sleep(300 * time.Millisecond)
        fmt.Println("goroutine 1: trying to lock muB")
        muB.Lock()
        muB.Unlock()
        muA.Unlock()
    }()
    go func() {
        defer wg.Done()
        fmt.Println("goroutine 2: locked muB")
        muB.Lock()
        time.Sleep(300 * time.Millisecond)
        fmt.Println("goroutine 2: trying to lock muA")
        muA.Lock()
        muA.Unlock()
        muB.Unlock()
    }()
    wg.Wait()
    fmt.Println("this line never prints; deadlock occurred")
}
\end{lstlisting}
\end{latin}

\noindent
اگر این فایل را اجرا کنیم ما در خروجی ابتدا می‌بینیم
\begin{enumerate}
    \item «\lr{goroutine 1: locked muA}» که نشان می‌دهد \lr{goroutine} نخست عملیات \lr{Lock} روی \lr{muA} را انجام داده است
    \item پس از مکث کوتاه «\lr{goroutine 2: locked muB}» که یعنی \lr{goroutine} دوم قفل \lr{muB} را به دست آورده است
    \item سپس \lr{goroutine} اول تلاش می‌کند قفل \lr{muB} را بگیرد اما چون \lr{muB} در اختیار \lr{goroutine} دوم است \lr{goroutine} اول بلوکه می‌شود
    \item \lr{goroutine} دوم نیز پس از تاخیر خودش به سراغ قفل \lr{muA} می‌رود اما \lr{muA} در اختیار \lr{goroutine} اول قرار دارد بنابراین او هم بلوکه می‌شود
    \item در نتیجه هر دو \lr{goroutine} برای همیشه منتظر آزاد شدن قفل دوم باقی می‌مانند و دستور \lr{wg.Wait()} هیچ‌گاه خاتمه پیدا نمی‌کند جمله پایانی نیز نمایش داده نمی‌شود
\end{enumerate}

\bigskip

\textbf{مثال دوم : دو کانال بدون بافر:}

در این مثال دو کانال بدون بافر (\lr{unbuffered}) با نام‌های \lr{ch1} و \lr{ch2} ایجاد کرده‌ایم دو \lr{goroutine} موازی اجرا می‌شوند \lr{goroutine} اول عدد \lr{1} را به \lr{ch1} می‌فرستد و سپس منتظر دریافت از \lr{ch2} می‌شود \lr{goroutine} دوم عدد \lr{2} را به \lr{ch2} می‌فرستد و سپس منتظر دریافت از \lr{ch1} می‌شود از آنجا که هر دو کانال بدون بافر هستند عملیات ارسال (\lr{send}) در هر کانال بلوکه می‌شود تا طرف مقابل (\lr{receive}) عمل کند در نتیجه هر دو \lr{goroutine} در وضعیت بن‌بست قرار می‌گیرند و برنامه قفل می‌شود

\medskip

\begin{latin}
\begin{lstlisting}[language=Go]
func main() {
    ch1 := make(chan int)
    ch2 := make(chan int)
    go func() {
        fmt.Println("goroutine 1: send 1 to ch1")
        ch1 <- 1
        fmt.Println("goroutine 1: waiting to receive from ch2")
        <-ch2
        fmt.Println("goroutine 1: if this prints, no deadlock")
    }()
    go func() {
        fmt.Println("goroutine 2: send 2 to ch2")
        ch2 <- 2
        fmt.Println("goroutine 2: waiting to receive from ch1")
        <-ch1
        fmt.Println("goroutine 2: if this prints, no deadlock")
    }()
    select {}
}
\end{lstlisting}
\end{latin}

\noindent
اگر این فایل را اجرا کنیم ما در خروجی خواهیم دید
\begin{enumerate}
    \item «\lr{goroutine 1: send 1 to ch1}» و «\lr{goroutine 2: send 2 to ch2}» تقریبا هم‌زمان چاپ می‌شوند
    \item \lr{goroutine} اول پس از ارسال به \lr{ch1} بلوکه می‌شود تا \lr{goroutine} دوم داده را از \lr{ch1} دریافت کند اما \lr{goroutine} دوم هنوز در مرحله ارسال به \lr{ch2} بلوکه است
    \item \lr{goroutine} دوم پس از ارسال به \lr{ch2} بلوکه می‌شود تا \lr{goroutine} اول داده را از \lr{ch2} دریافت کند در حالی که \lr{goroutine} اول هنوز در ارسال به \lr{ch1} بلوکه مانده است
    \item بنابراین هر دو \lr{goroutine} در وضعیت بلوکه باقی می‌مانند و هیچ‌یک از چاپ‌های «\lr{if this prints, no deadlock}» را مشاهده نمی‌کنیم
    \item همچنین دستور \lr{select \{\}} هیچ‌گاه خاتمه نمی‌یابد و برنامه تا ابد در حالت بن‌بست باقی می‌ماند
\end{enumerate}

\bigskip

در هر دو مثال وقتی فایل‌ها را اجرا کنیم می‌بینیم که پس از چند خط خروجی اولیه بن‌بست رخ داده و برنامه دیگر پیشرفتی ندارد 



\section*{شماره سوال 2}

می‌دانیم که می‌توانیم روی یک اتصال \lr{TCP} یک دنباله از بایت‌ها را ارسال کنیم که تعداد آن‌ها دلخواه است. فرض کنید بخواهیم بر روی یک اتصال \lr{TCP} چندین درخواست و پاسخ را ارسال کنیم. چگونه می‌توان چنین پروتکلی را طراحی کنیم؟

وقتی می‌خواهیم روی یک اتصال \lr{TCP} چندین درخواست و پاسخ را پشت‌سرهم بفرستیم، باید اول بدانیم که \lr{TCP} خودش فقط یک جریان بایت (\lr{byte stream}) ارائه می‌کند و هیچ مرزی بین پیام‌های مختلف قائل نیست. یعنی اگر بدون سازوکار خاصی شروع کنیم به فرستادن چند پیام پی‌درپی، سمت گیرنده مقابل نمی‌فهمد پیام اول کجا تمام شده و پیام بعدی از کجا شروع شده. برای رفع این مشکل، باید خودمان کاری کنیم تا هر پیام یک \lr{frame} داشته باشد؛

یک روش مرسوم این است که قبل از هر پیام، یک \lr{length-prefix} بفرستیم. فرض کنید می‌خواهیم یک \lr{payload} دلخواه (مثلا یک رشته \lr{JSON} یا داده باینری) را ارسال کنیم. ابتدا طول آن \lr{payload} را به صورت یک عدد ثابت‌طول (مثلا ۴ باایت با ترتیب بایت \lr{Big-Endian} یا \lr{Little-Endian}) می‌فرستیم و بلافاصله بعدش همان تعداد باایت پیام را می‌فرستیم. با این روش گیرنده اولین چهار باایت را می‌خواند، این عدد را تفسیر می‌کند و می‌داند باید چند باایت بعدی را بخواند تا پیام کامل شود. وقتی آن \lr{N} باایت را کامل گرفت، دوباره می‌چرخد سراغ چهار باایت بعدی به عنوان طولِ پیام دوم و الی آخر. 

در عمل معمولا برای هر پیام علاوه بر آن فیلد طول، یک فیلد شناسه هم می‌گذارند (مثلا عدد ۴ بایتی به \lr{Big-Endian} یا \lr{Little-Endian} می‌نویسند) تا اگر چند درخواست هم‌زمان داشته باشیم و پاسخ‌ها به ترتیب دلخواه برسند، بتوانیم هر پاسخ را به درخواست مربوطش وصل کنیم. در این حالت ساختار پیام می‌شود چیزی مثل:
\[
    \underbrace{\text{\lr{ID (4 bytes)}}}_{\text{شناسه}} \quad
    \underbrace{\text{\lr{LEN (4 bytes)}}}_{\text{طول \lr{Payload}}} \quad
    \underbrace{\text{\lr{Payload} (\lr{LEN} \lr{bytes})}}_{\text{محتوای اصلی}}
\]
گیرنده اول هشت باایت را می‌خواند، فرض می‌کند عدد اول (مثلا ۱۷) شناسه پیام است و عدد دوم (مثلا ۱۲۳) طول \lr{payload} را نشان می‌دهد. سپس باید ۱۲۳ باایت بعدی را بخواند تا \lr{Payload} را کامل دریافت کند. وقتی آن ۱۲۳ باایت را خواند، یک پیام کامل دارد که می‌تواند بر اساس \lr{ID} یا نوع پیام آن را پردازش کند. همین‌طور اگر بخواهیم ده‌ها پیام را پی‌درپی ارسال کنیم، کافی است همین الگو را برای هر پیام تکرار کنیم: اول هدر هشت بایتی (\lr{ID} و \lr{LEN}) و بعد \lr{Payload} به تعداد \lr{LEN} بایت، و سپس هدر بعدی و \lr{Payload} بعدی و الی آخر.

یک روش دیگر هم استفاده از جداساز (\lr{delimiter}) است؛ یعنی به جای اینکه حتما طول پیام را قبلش بفرستیم، یک الگوی خاص تعریف می‌کنیم که نشان دهد «این نقطه انتهای پیام است». مثلا در بسیاری از پروتکل‌های متنی از دنباله دو بایت \lr{\textbackslash r\textbackslash n} به عنوان پایان پیام استفاده می‌کنند. در این حالت هر پیام را سریال می‌کنیم (مثلا \lr{JSON} می‌شود رشته \lr{UTF-8}) و پس از آن کاراکترهای \lr{\textbackslash r\textbackslash n} می‌فرستیم. گیرنده یک بافر درونی نگه می‌دارد و هر بار تعدادی بایت از \lr{TCP} می‌خواند و به انتهای بافر اضافه می‌کند. وقتی در بافر دنباله \lr{\textbackslash r\textbackslash n} پیدا شد، می‌فهمد که تا آن نقطه یک پیام کامل دریافت کرده است. سپس آن‌قدر بایت قبل از \lr{\textbackslash r\textbackslash n} را جدا می‌کند و برای پردازش تحویل می‌دهد و باقی بافر را (پس از آن \lr{\textbackslash r\textbackslash n}) نگه می‌دارد برای پیام‌های بعدی. در این روش باید مراقب باشیم خود \lr{Payload} حاوی همان دنباله جداکننده نباشد؛ در غیر این صورت قبل از اینکه پیام به پایان رسیده، اشتباها گیرنده فکّر می‌کند پایان پیام است. برای همین اگر احتمال دارد در \lr{payload} از همان الگو استفاده شود، معمولا باید \lr{payload} را قبل از ارسال \lr{encode} کنیم یا از دنباله‌ای استفاده کنیم که مطمئن هستیم در داده اصلی تکرار نمی‌شود.

در عمل اگر از یک پروتکل متن‌محور شبیه \lr{HTTP/1.1} استفاده کنیم، در ابتدا تعدادی خط هدر داریم که با \lr{\textbackslash r\textbackslash n} پایان می‌یابند و پایان بخش هدرها با یک خط خالی (دو بار \lr{\textbackslash r\textbackslash n} پشت‌سرهم) مشخص می‌شود. اگر بخواهند بدنه \lr{HTTP} بفرستند، معمولا از هدر \lr{Content-Length: <number>} استفاده می‌کنند تا گیرنده بداند باید دقیقا چند بایت را پس از هدرها به عنوان بدنه بخواند. این ترکیب در \lr{HTTP}، هم از \lr{delimiter} خطی (برای پایان هدر) و هم از پیش‌عنوان طول (برای پایان \lr{body}) استفاده می‌کند.

یک نکته خیلی مهم دیگر این است که وقتی قصد داریم چند درخواست و پاسخ را روی یک کانکشن \lr{TCP} جابه‌جا کنیم، معمولا حالت همزمانی (\lr{concurrency}) را هم مدنظر داریم. فرض کنید چند \lr{goroutine} در کلاینت داریم که می‌خواهند درخواست بفرستند. بهتر است یک \lr{goroutine} واحد مسئول نوشتن روی سوکت باشد و همه \lr{goroutine} های دیگر وقتی می‌خواهند درخواست بفرستند، پیامشان را در یک کانال داخلی (\lr{Go channel}) قرار دهند. آن \lr{writer goroutine} مرتب از آن کانال می‌خواند و \lr{Write} می‌کند روی سوکت، تا مطمئن باشیم دو \lr{goroutine} همزمان وارد \lr{Write} نشوند و تداخل پیش نیاید. از طرف دیگر، برای خواندن هم یک \lr{goroutine} واحد داریم که دائم \lr{Read} می‌کند؛ داده‌ها را در بافر داخلی می‌چیند، تا وقتی که یک پیام کامل نرسیده صبر می‌کند، و وقتی پیام کامل شد، آن را تحویل یک کانال دیگر یا تابع پردازش اصلی می‌دهد. با این ساختار ساده، هیچ‌وقت چند \lr{goroutine} به طور هم‌زمان روی سوکت نمی‌خوانند یا نمی‌نویسند و احتمال تداخل یا بلوک شدن ناخواسته کاهش می‌یابد.

اگر قرار باشد سرور همزمان چندین درخواست را پاسخ بدهد و ترتیب پاسخ‌ها الزاما مثل ترتیب درخواست‌ها نباشد (مثلا درخواستی ممکن است پردازشش طول بکشد و بعد پاسخش دیر برسد)، حتما باید از شناسه (\lr{ID}) استفاده کنیم. یعنی وقتی هر \lr{goroutine}‌ای درخواستش را می‌سازد، یک \lr{ID} یکتا انتخاب می‌کند و داخل هدر پیام (مثلا هم‌زمان با طول) می‌فرستد. سمت سرور که پاسخ را آماده می‌کند، همان \lr{ID} را در پاسخ می‌گذارد. هنگام دریافت جواب، گیرنده می‌تواند بر اساس \lr{ID} بفهمد این پاسخ مربوط به کدام درخواست بوده و آن را به \lr{goroutine} درست تحویل بدهد. اگر ترتیب \lr{FIFO} برایمان کافی باشد (یعنی مطمئنیم ترتیب پاسخ دقیقا مثل ترتیب درخواست خواهد بود)، ممکن است نیازی به \lr{ID} نداشته باشیم، اما اغلب مواقع که چند درخواست هم‌زمان داریم و پاسخ‌ها ممکن است به ترتیب دلخواه برسند، \lr{ID} ضروری است تا ارتباط برقرار باشد.

در \lr{Protocol Buffers} (\lr{protobuf})، زیرمجموعه‌ای از \lr{IDL} ارائه می‌شود که با فایل‌های \lr{.proto} تعریف می‌شود. کامپایلر \lr{protoc} از این فایل‌ها \lr{code} تولید می‌کند تا کلاینت و سرور در زبان‌های مختلف (مثلاً \lr{Go}، \lr{Java}، \lr{Python}) طبق قرارداد مشترک عمل کنند.

\textbf{۱. تعریف \lr{message}ها:}

این کد دو \lr{message} تعریف کرده که مانند \lr{struct} در زبان‌های برنامه‌نویسی عمل می‌کنند. هر \lr{message} برای تعریف \lr{data type} جدید استفاده می‌شود. در هر \lr{message} می‌توانیم \lr{required} یا \lr{optional} بودن فیلدها را مشخص کنیم. عدد بعد از علامت مساوی شماره \lr{field} است. شماره \lr{field} یک عدد صحیح مثبت است که در فرایند \lr{serialization} پیام به \lr{binary}، برای شناسایی فیلدهای مختلف به‌کار می‌رود و امکان \lr{forward} و \lr{backward compatibility} را فراهم می‌کند. 

\medskip

\noindent
\textbf{۱.۱. \lr{message PaymentRequest}}  
این \lr{message} ساختار \lr{PaymentRequest} را تعریف می‌کند که برای ارسال اطلاعات پرداخت از کلاینت به سرور استفاده می‌شود.  
\begin{itemize}
    \item \lr{message Card}  
    \begin{itemize}
        \item \lr{required string cardNumber = 1;}  
        این فیلد شماره کارت را به شکل یک رشته \lr{UTF-8} نگه می‌دارد. در \lr{protobuf} وقتی \lr{required} مشخص است، ارسال این فیلد اجباری است و اگر در پیام نباشد در زمان \lr{(de)serialization} خطا رخ می‌دهد.  
        \item \lr{optional int32 expiryMonth = 2;}  
        تاریخ انقضای کارت به شکل عدد صحیح ۳۲ بیتی که ماه را مشخص می‌کند (از ۱ تا ۱۲). به دلیل \lr{optional} بودن، در صورت نبودن این فیلد، مقدار پیش‌فرض صفر (۰) استفاده می‌شود؛ اما کلاینت می‌تواند این فیلد را حذف کند اگر نیازی به آن نداشته باشد.  
        \item \lr{optional int32 expiryYear = 3;}  
        سال انقضای کارت به صورت عدد صحیح ۳۲ بیتی. مشابه \lr{expiryMonth}، اگر ارسال نشود مقدار پیش‌فرض صفر خواهد بود.  
        \item \lr{optional int32 CVC = 4;}  
        کد امنیتی (\lr{CVC}) کارت به صورت عدد صحیح ۳۲ بیتی. اختیاری است و می‌توان آن را حذف کرد مگر اینکه منطق کسب‌وکار نیاز به آن داشته باشد.  
    \end{itemize}
    \item \lr{enum Currency \{ GBP = 1; USD = 2; \}}  
    این بخش یک \lr{enum} به نام \lr{Currency} تعریف می‌کند. \lr{enum} در \lr{protobuf} فهرستی از مقادیر نام‌گذاری‌شده است که هر کدام یک \lr{integer} را به خود اختصاص می‌دهند.  
    \item \lr{required Card card = 1;}  
    این فیلد از نوع \lr{Card} است که همان \lr{message Card} تعریف‌شده در بالا است. به این ترتیب ساختار تودرتو داریم. اگر کلاینت این فیلد را ارسال نکند، خطای \lr{required field missing} رخ می‌دهد.  
    \item \lr{required int64 amount = 2;}  
    این فیلد مبلغ پرداختی (\lr{amount}) را به صورت عدد صحیح ۶۴ بیتی نگه می‌دارد. \lr{amount} باید مقدار دلخواهی به عنوان مبلغ باشد و ارسال آن اجباری است؛ در صورت نبودن مقدار، پیام اعتبارسنجی نمی‌گذرد.  
    \item \lr{required Currency currency = 3;}  
    این فیلد نشان می‌دهد مبلغ بالا به چه ارزی پرداخت می‌شود؛ از نوع \lr{Currency} است که در بالا تعریف شد. اگر ارسال این فیلد نباشد، خطا رخ می‌دهد.  
\end{itemize}

\medskip

\noindent
\textbf{۱.۲. \lr{message PaymentStatus}}  
این \lr{message} ساختار \lr{PaymentStatus} را تعریف می‌کند که سرور پس از دریافت \lr{PaymentRequest} و پردازش پرداخت، باز می‌گرداند:  
\begin{itemize}
    \item \lr{required bool success = 1;}  
    یک مقدار \lr{boolean} است که نشان می‌دهد آیا پرداخت موفق بوده (\lr{true}) یا ناموفق (\lr{false}). چون \lr{required} است حتماً باید ارسال شود و اگر در پیام نباشد، خطای \lr{required field missing} رخ می‌دهد.  
    \item \lr{optional string errorMessage = 2;}  
    در صورتی که \lr{success} برابر \lr{false} باشد، می‌توان یک \lr{string} شامل توضیح خطا ارسال کرد. اگر \lr{success} \lr{true} باشد، معمولاً این فیلد حذف می‌شود. در حالت \lr{optional}، اگر ارسال نشود مقدار پیش‌فرض رشته خالی در نظر گرفته می‌شود.  
\end{itemize}

\bigskip

\noindent
\textbf{۲. تعریف \lr{service}:}

در بخش \lr{service} ما یک رابط \lr{RPC} تعریف می‌کنیم. \lr{service} در \lr{protobuf} معادل گروهی از متدهای \lr{RPC} است که کلاینت می‌تواند روی سرور فراخوانی کند.  

\begin{itemize}
    \item \lr{service PaymentService \{\}}  
    ابتدا یک \lr{service} به نام \lr{PaymentService} ایجاد می‌کنیم. این سرویس مجموعه‌ای از متدهای \lr{RPC} را تعریف می‌کند.  
    \item \lr{rpc ProcessPayment(PaymentRequest) returns (PaymentStatus) \{\}}  
    این خط یک متد \lr{RPC} به نام \lr{ProcessPayment} تعریف می‌کند که از نوع \lr{unary RPC} است (یک درخواست و یک پاسخ).  
    ساختار این فراخوانی به این شکل است که کلاینت ابتدا یک شیء \lr{PaymentRequest} را سریالیزه می‌کند (طبق تعریف فیلدها و شماره \lr{field}) و روی \lr{TCP} به سرور می‌فرستد. سرور پیام دریافت‌شده را دسیریالیزه می‌کند، منطق کسب‌وکار پرداخت را اجرا می‌کند (مثل اعتبارسنجی کارت، برقراری ارتباط با درگاه بانکی و ...) و در نهایت یک شیء \lr{PaymentStatus} تولید می‌کند. این شیء را سریالیزه کرده و به کلاینت برمی‌گرداند.  
\end{itemize}

\begin{itemize}
    \item در \lr{Protocol Buffers} هر فیلد یک شماره یکتا (\lr{field number}) دارد که در مشخصه \lr{= 1}، \lr{= 2} و غیره دیده می‌شود. این عدد در ساختار باینری پیام به‌صورت \lr{tag} ذخیره می‌شود که شامل شماره فیلد و \lr{wire type} است. \lr{wire type} نشان می‌دهد داده فیلد چگونه کدگذاری می‌شود .  
    \item وقتی سرور یا کلاینت پیام را دسیریالیزه می‌کند، ابتدا \lr{tag} را مطالعه می‌کند: \lr{field number} را استخراج می‌کند و سپس بسته به نوع داده بایت‌های بعدی را می‌خواند.  
    \item \lr{required} به این معنی است که پیغام باید حتماً آن فیلد را داشته باشد؛ اگر نباشد، در حین دسیریالیزه کردن خطا می‌دهد. \lr{optional} به این معنی است که اگر فیلد ارسال نشود، مقدار پیش‌فرض (صفر برای عددها، رشته خالی برای \lr{string} و \lr{false} برای \lr{bool}) در نظر گرفته می‌شود.  
    \item \lr{enum} نیز به \lr{protobuf} اجازه می‌دهد یک مجموعه محدود از مقادیر صحیح را با نام‌های خوانا تعریف کند. در اینجا \lr{Currency} به جای ارسال یک \lr{int32} آزاد، دو مقدار \lr{GBP} یا \lr{USD} را محدود می‌کند. این باعث جلوگیری از خطا در کد کلاینت و سرور می‌شود که فقط مقادیر مجاز را ارسال کنند.  
\end{itemize}

\section*{شماره سوال 4}

دو ارتش از انسان ها و ا ِلف ها شهر موردور را محاصره كرده اند و قصد حمله به آن را دارند. شما فرماندهي ارتش انسان ها
را به عهده داريد. مواضع دفاعي موردور نسبتاً خوب است و اگر يك ارتش به تنهايي به آن حمله كند حتماً شكست مي خورد. اما
اگر دو ارتش به آن حمله كنند موردور سقوط مي كند. تنها راه ارتباطي دو فرمانده ارسال يك سرباز به عنوان پيام رسان است كه
البته ممكن است در بين راه دستگير شود و پيام را به مقصد نرساند. براي هماهنگي با فرمانده ديگر چه متني را پيشنهاد مي كنيد
و استراتژي شما براي غلبه بر موردور چيست؟ 

\noindent
\textbf{۱. عدم‌قطعیت در هماهنگی، چرا هیچ راه قطعی‌ای وجود ندارد؟}

مساله آنجا آشکار می‌شود که تنها راه ارتباطی بین دو فرمانده فرستادن یک سرباز (\lr{messenger}) است و ممکن است در مسیر اسیر شود و پیام به مقصد نرسد. در نتیجه:
\begin{itemize}
    \item هر بار که یک پیام ارسال می‌شود، احتمال \(p\) وجود دارد که آن پیام اسیر شود و هیچ تضمینی برای رسیدن ندارد.
    \item حتی اگر فرمانده انسان‌ها پیام «حمله در چهارشنبه هفته آینده با طلوع آفتاب» را بفرستد و فرمانده الف‌ها پاسخ تایید بدهد، همان پاسخ تایید هم ممکن است در مسیر گم شود.
    \item فرض کنید دو مرحله تایید داریم: اول پیام اصلی، سپس پاسخِ الف‌ها و بعد باز تایید نهایی انسان‌ها. اگر «آخرین تایید» اسیر شود، یک طرف می‌پندارد هم‌زمانی رخ داده و طبق برنامه عمل می‌کند، و طرف دیگر معتقد است هنوز هماهنگی کامل نشده و حمله را به تاخیر می‌اندازد؛ در نتیجه حمله هم‌زمان منتفی می‌شود.
    \item این ساختارِ \lr{unreliable communication channel} یعنی هیچ پیامی نمی‌تواند تضمین کند که به مقصد رسیده و دریافت‌ شده‌است. حتی اگر تعداد تاییدها را زیاد کنیم، باز هم احتمال نرسیدنِ «آخرین بسته» (پیام یا تایید) همیشه بالاست و هرگز به صفر نمی‌رسد.
    \item از این رو، اثبات می‌شود که هیچ پروتکلی در چنین شرایطی نمی‌تواند هماهنگی قطعی فراهم کند؛ فقط می‌توان احتمال خطا را به دلخواه کوچک کرد ولی صفر نه.
\end{itemize}

\noindent
به همین دلیل تنها راه مطمئن برای هماهنگی، تکیه بر یک مرجع از پیش توافق‌شده (مثلاً زمان مشخص بر اساس تقویم) است که نیازی به تاییدِ پیام‌‌رسان لحظه‌ای نداشته باشد.

\bigskip

\noindent
\textbf{۲. استراتژی‌های پیشنهادی برای هماهنگی}

1. \textbf{تعیین زمان از پیش بدون نیاز به تایید:}  
   اگر بتوانیم از قبل یک زمان دقیق را توافق کنیم مثلاً «چهارشنبه هفته آینده با طلوع آفتاب حمله کنیم  دیگر نیازی به ردوبدل مکرر پیام نیست. در این حالت، حتی اگر پیام اول (اطلاع از زمان) به فرمانده الف‌ها نرسد، هر دو طبق توافق از پیش انجام‌شده عمل می‌کنند. وقتی آن روز فرا برسد، هر دو هم‌زمان هجوم خواهند برد و موردور بدون دفاع هم‌زمان، شکست می‌خورد.

2. \textbf{ارسال و دریافت چند مرحله‌ای پیام با تایید دریافت (\lr{ACK}):}  
   اگر بخواهیم احتمال هماهنگی را بیشتر کنیم، می‌توانیم چند مرحله پیام و تایید (\lr{ACK}) ترتیب دهیم:
   \begin{enumerate}
       \item فرمانده انسان‌ها پیامِ «حمله در چهارشنبه هفته آینده با طلوع آفتاب» را می‌فرستد.
       \item اگر فرمانده الف‌ها آن پیام را دریافت کند، با پیام «پیغام شما دریافت شد؛ ما نیز چهارشنبه هفته آینده با طلوع آفتاب حمله می‌کنیم» پاسخ می‌دهد.
       \item اگر این پاسخ نرسد، انسان‌ها دوباره همان پیام را ارسال می‌کنند و منتظر پاسخ تایید می‌مانند.  
       \item بعد از اینکه یک بار پاسخ تایید رسید، انسان‌ها یک بار دیگر تاییدِ دریافتِ آن پاسخ را برای الف‌ها می‌فرستند تا مطمئن شوند الف‌ها می‌دانند پیامشان هم رسیده.  
       \item اگر تا دو روز قبل از زمان مقرر دست‌کم یکی از این تاییدها ردوبدل شده باشد، دو طرف احتمال خطای نهایی را به شدت کاهش داده‌اند و می‌توانند بدون ارسال پیام بیشتر، بر اساس زمان از پیش تعیین‌شده حمله کنند.
   \end{enumerate}
   در این روش، هرچقدر تعداد مراحل تایید بیشتر باشد، احتمال نرسیدن آن «آخرین تایید» پایین‌تر می‌رود اما هرگز به صفر نمی‌رسد.

3. \textbf{متن پیشنهادی برای پیام‌ها:}  
   \begin{itemize}
   \item {پیام اول از انسان به الف‌ها:}  
        «ما (فرمانده انسان‌ها) تصمیم گرفته‌ایم چهارشنبه هفته آینده با طلوع آفتاب از جبهه شمال شرقی حمله کنیم. اگر این پیام به شما رسید، با عبارت «پیغام شما دریافت شد؛ ما چهارشنبه هفته آینده از جبهه شمال غربی حمله می‌کنیم» پاسخ دهید. در غیر این صورت، 

   \item {پاسخ اول از الف‌ها به انسان‌ها (تایید دریافت):}  
        «پیغام شما دریافت شد؛ ما چهارشنبه هفته آینده از جبهه شمال غربی حمله می‌کنیم.»

   \item {تایید نهایی از سوی انسان‌ها به الف‌ها:}  
        «پیغام تایید شما دریافت شد؛ هم‌زمان با شما در چهارشنبه هفته آینده با طلوع آفتاب از جبهه شمال شرقی حمله می‌کنیم.»
   \end{itemize}
   اگر هیچ‌یک از تاییدها تا دو روز قبل از حمله نرسید، هیچ‌کدام از ارتش‌ها دست به حمله تنها نمی‌زنند و منتظر موقعیت بعدی می‌شوند.

\bigskip

با ترکیب روش اول (تعیین زمان از پیش) و روش دوم (ارسال چند مرحله‌ای پیام با \lr{ACK})، می‌توان احتمال هماهنگی هم‌زمان را تا حد بسیار زیادی بهبود بخشید. اگر حداقل یک بار تایید تا دو روز قبل از حمله ردوبدل شود، هر دو فرمانده مطمئن می‌شوند طرف مقابل زمان حمله را می‌داند و حتماً هماهنگ خواهد بود. در نتیجه در لحظه طلوع آفتاب، سپاه انسان‌ها از شمال شرقی و سپاه الف‌ها از شمال غربی هم‌زمان هجوم برده و قلعه موردور با دفاعِ یکپارچه هم‌زمان توان مقاومت نخواهد داشت و سقوط خواهد کرد.

\noindent
\textbf{الف: محاسبه پیچیدگی الگوریتم}  
در این الگوریتم ابتدا گره اولیه (که پیام را ایجاد کرده) پیام را به همه \(N-1\) گره دیگر (و می‌تواند خودش را هم بشمارد اما در عمل ارسال به خودش بی‌معنی است) ارسال می‌کند. بنابراین در مرحله اول تعداد \(N-1\) پیام فرستاده می‌شود. سپس هر یک از آن \(N-1\) گره‌ای که پیام را دریافت کرده، برای بار دوم پیام را به همه \(N-1\) گره دیگر (شامل خود ارسال‌کننده اولیه یا سایر گره‌ها) می‌فرستند. عملاً اگر همه فرض کنیم پیام را دریافت کنند، در مرحله دوم تقریباً \((N-1)\times(N-1) = (N-1)^2\) ارسال داریم که نزدیک به \(N^2\) است. به عبارت دیگر، در وضعیت عمومی که همه گره‌ها پیام را دریافت می‌کنند و دوباره منتشر می‌کنند، کل تعداد پیام‌های ارسالی در شبکه برابر  
\[
(N-1) \;+\; (N-1)^2 \;\approx\; N^2  
\]  
خواهد بود. در نتیجه پیچیدگی این الگوریتم از نظر تعداد پیام‌های ارسالی برابر \(O(N^2)\) است.

\bigskip

\noindent
\textbf{ب: بهبود الگوریتم از نظر سربار پیام}  
در الگوریتم فوق ممکن است پیامِ یکسان بارها به یک گره برسد و آن گره نیز دوباره آن را به همه گره‌ها ارسال کند؛ در نتیجه تعداد پیام‌ها بسیار زیاد می‌شود. برای کاهش این سربار می‌توان از روش‌های زیر استفاده کرد:

1. نگهداری شناسه پیام‌های منتشرشده:  
   هر گره یک ساختار داده (مثلاً \lr{list} یا \lr{set}) از شناسه‌های پیام‌هایی که قبلاً منتشر کرده نگه می‌دارد. وقتی یک پیام جدید دریافت می‌کند، ابتدا شماره \lr{message ID} را چک می‌کند. اگر قبلاً همین شناسه را در لیست داشته باشد، به این معناست که پیش‌تر پیام را منتشر کرده و نباید دوباره آن را بفرستد. در غیر این صورت شناسه را به لیست اضافه می‌کند و پیام را تنها یک بار برای گره‌هایی که هنوز پیام را دریافت نکرده‌اند ارسال می‌کند. این کار جلوی ارسال مکرر پیام‌های تکراری را می‌گیرد و تقریباً تضمین می‌کند هر گره حداکثر یک‌بار آن پیام را منتشر کند.

2. استفاده از لیست دریافت‌کنندگان یا \lr{forwarding list}:  
   می‌توان در هر پیام یک \lr{forwarding list} از شناسه گره‌هایی که تاکنون پیام را دریافت کرده‌اند گنجاند. وقتی گره‌ای پیام را دریافت می‌کند، اول این لیست را بررسی می‌کند و تنها به آن گره‌هایی ارسال می‌کند که شناسه‌شان در لیست نیست. سپس شناسه خودش را به لیست اضافه کرده و پیام را برای باقی گره‌های لازم می‌فرستد. به این ترتیب یک پیام فقط به زیرمجموعه‌های جدید ارسال می‌شود و جلوی ارسال‌های اضافی به گره‌هایی که پیش‌تر پیام را داشته‌اند گرفته می‌شود.

3. استفاده از ساختار درخت پوششی (\lr{spanning tree}):  
   اگر بتوانیم یک \lr{spanning tree} مستقل از وضعیت فعلی شبکه بسازیم، هر گره پیام را تنها به فرزندانش در درخت می‌فرستد. در نتیجه تعداد کل پیام‌های ارسالی تقریباً \(N-1\) خواهد بود (هر لبه درخت یک‌بار پیام را عبور می‌دهد). با وجود احتمال توقف موقت برخی گره‌ها، اگر درخت پوششی \lr{self-healing} داشته باشیم یا چندین درخت پوششی متفاوت تعریف کنیم، می‌توانیم هم قابلیت اتکای بالا و هم سربار پایین را تا حد زیادی تضمین کنیم. در این حالت دیگر نیازی نیست هر گره پیام را به همه همسایه‌ها بفرستد، بلکه فقط به آن دسته از همسایه‌هایی ارسال می‌شود که زیرشاخه آن‌ها در درخت قرار دارد.

4. گسترش با الگوریتم \lr{gossip-based broadcast}:  
   در این روش هر گره پیام را تنها به تعدادی از همسایه‌های تصادفی (به‌عنوان مثال \(\log N\) همسایه) ارسال می‌کند. سپس هر همسایه با احتمال از پیش تعیین‌شده پیام را به دیگران مخابره می‌کند. مزیت این روش این است که تعداد ارسال‌ها به صورت احتمالی \(O(N \log N)\) یا حتی \(O(N)\) خواهد بود در حالی که با تکرارهای کافی احتمال آنکه پیام به تمام گره‌ها برسد بسیار بالا می‌رود. تنها نکته این است که تضمین قطعی برای رسیدن پیام به همه گره‌ها از بین می‌رود و به جای آن احتمال بسیار بالا (مثلاً \(1 - \frac{1}{N}\)) را به دست می‌آوریم. در محیط‌هایی که می‌توان سربار کمی را پذیرفت تا احتمال شکست را اساساً صفر کرد، این روش بسیار کارا است.

با اعمال هر یک از این بهینه‌سازی‌ها (نگهداری \lr{message ID}، استفاده از \lr{forwarding list}، درخت پوششی یا انتشار به سبک \lr{gossip-based broadcast}) می‌توان سربار پیام در الگوریتم همه‌پخشی از \(O(N^2)\) به تقریباً \(O(N)\) یا \(O(N \log N)\) کاهش داد و در عین حال تا حد زیادی تضمین کرد که پیام در نهایت به همه گره‌های فعال می‌رسد یا لااقل تشخیص داده می‌شود که پیام کامل به همه نرسیده است.


\section*{پاسخ سوال 6}
\noindent
یک همه‌پخشی را ترتیب کلی \lr{FIFO} می‌گوییم اگر دو خاصیت زیر را داشته باشد:
\begin{enumerate}
    \item اگر یک گره دو پیام را یکی بعد از دیگری از طریق همه‌پخشی ارسال کند، آن‌گاه پیام‌ها به همان ترتیب همه‌پخشی به گره‌های دیگر تحویل می‌شوند.
    \item اگر پیام \lr{m1} در یک گره قبل از پیام \lr{m2} تحویل شود، آنگاه ترتیب تحویل این دو پیام در تمام گره‌های دیگر به همین صورت است.
\end{enumerate}
شبه‌کدی بنویسید که یک همه‌پخشی ترتیب کلی \lr{FIFO} را پیاده‌سازی کند. فرض کنید هر گره یک شناسه یکتا دارد و هر گره شناسه تمام گره‌های دیگر را می‌داند.


برای برآورده کردن این قیود، از ایده \lr{logical clock} مستقل برای هر \lr{node} استفاده می‌کنیم و یک شمارنده عددی می‌سازیم که با هر ارسال یک \lr{message}، مقدارش یکی افزایش یابد. این مقدار به‌عنوان \lr{sequence number} به همراه خودِ \lr{message} برای باقی \lr{node}ها فرستاده می‌شود. وقتی هر \lr{node} \lr{message}ای را دریافت می‌کند، آن را در یک \lr{buffer} موقتی قرار می‌دهد تا زمانی که تمام \lr{message}های با \lr{sequence number}‌های کوچک‌تر از همان \lr{sender} تحویل داده شوند. تنها پس از آن اجازه می‌دهیم آن \lr{message} \lr{deliver} شود. با این روش:

\begin{itemize}
    \item اگر \lr{node X} دو \lr{message} به ترتیب با شماره‌های \(s\) و \(s+1\) ارسال کرده، همه \lr{node}های دیگر ابتدا \(\;s\) را دریافت و تحویل می‌دهند و بعد \(\;s+1\) را، بنابراین ترتیب ارسال در هر نقطه ثابت می‌ماند (خاصیت ۱).  
    \item اگر \lr{node Y} هر \lr{message m1} را با شماره \(s_1\) زودتر از \lr{message m2} با شماره \(s_2\) دریافت کند (یعنی \(s_1 < s_2\))، همه \lr{node}های دیگر هم به ترتیب منطقیِ \lr{sequence number} همین ترتیب را خواهند داشت (خاصیت ۲).  
\end{itemize}

در ادامه متغیرهای موردنیاز و شبه‌کدِ کامل‌تر را می‌بینید.

\bigskip

\noindent
\textbf{متغیرها در هر \lr{node}:}  
\begin{itemize}
    \item \lr{clock} \(\leftarrow 0\)  
    \item \lr{buffer} \(\leftarrow \{\}\)  
    \item \lr{lastDelivered}[\lr{pid}] \(\leftarrow 0\) برای همه \lr{pid}  
    \item \lr{allNodes} \(\leftarrow\) لیست شناسه تمام \lr{node}ها  
\end{itemize}

\bigskip

\noindent
\textbf{شبه‌کد کامل‌تر:}

\begin{latin}
\begin{lstlisting}[language=Python]
def FIFO_Broadcast(msg):
    clock ← clock + 1
    seq ← clock
    for each q in allNodes:
        send (msg, seq, self_id) to q

def on_receive(msg, seq, sender_id):
    buffer.append((msg, seq, sender_id))
    try_deliver()

def try_deliver():
    repeat:
        delivered ← false
        for each (m, s, p) in buffer:
            if s == lastDelivered[p] + 1:
                deliver(m, p)
                lastDelivered[p] ← s
                buffer.remove((m, s, p))
                delivered ← true
    until not delivered

def deliver(m, p):
    print("Delivered message from", p, ":", m)
\end{lstlisting}
\end{latin}

\bigskip

\noindent
این الگوریتم تضمین می‌کند که:
\begin{itemize}
    \item هر \lr{node} با هر فراخوانی \lr{FIFO_Broadcast} یک \lr{sequence number} منحصربه‌فرد تولید کرده و ارسال می‌کند.
    \item در تابع \lr{try_deliver} تنها زمانی \lr{message} از \lr{buffer} جدا شده و \lr{deliver} می‌شود که همه \lr{message}های قبلی آن \lr{sender} دریافت و تحویل شده باشند.
\end{itemize}
